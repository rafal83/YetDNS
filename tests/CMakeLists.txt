cmake_minimum_required(VERSION 3.16)
project(yetdns_core_tests LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../tools/build_blocklist.py
    --output ${CMAKE_CURRENT_BINARY_DIR}/blocklist.test.bin ${CMAKE_CURRENT_SOURCE_DIR}/data/blocked.txt
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(core_tests core_tests.cpp ../main/dns_core.cpp)
target_include_directories(core_tests PRIVATE ../main)
target_compile_features(core_tests PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(core_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME core COMMAND core_tests ${CMAKE_CURRENT_BINARY_DIR}/blocklist.test.bin)
add_executable(list_tests list_builder_tests.cpp ../main/list_builder.cpp ../main/dns_core.cpp)
target_include_directories(list_tests PRIVATE ../main)

endif()
add_test(NAME lists COMMAND list_tests ${CMAKE_CURRENT_BINARY_DIR}/blocklist.test.bin)

    ${CMAKE_CURRENT_BINARY_DIR}/applications.txt COMMAND_ERROR_IS_FATAL ANY)
